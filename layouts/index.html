Smart Pointers at a glance

smart_pointers.cpp 
// unique_ptr — exclusive ownership
auto p1 = std::make_unique<Foo>();
auto p2 = std::move(p1);  // transfer ownership

// shared_ptr — shared ownership (ref-counted)
auto sp1 = std::make_shared<Foo>();
auto sp2 = sp1;  // use_count → 2

// weak_ptr — observe without owning
std::weak_ptr<Foo> wp = sp1;
if (auto tmp = wp.lock()) {
    // safe to use tmp here
}
